# =============================================================
#  PC side Python dependencies
# =============================================================
#  IMPORTANT: To properly install dlib / face_recognition on Windows,
#  the packages must be installed in a specific order. 
#  The easiest way is to use the setup script:
#
#     powershell -ExecutionPolicy Bypass -File pc\setup.ps1
#
#  This script creates a .venv and installs the following in order.
#  (Python 3.10+ and pip are sufficient; NO Conda. dlib-bin = no compilation.)
# -------------------------------------------------------------
numpy                    # numpy 2.x (2.2.6) tested: works fine with dlib-bin 20 + opencv 4.13
dlib-bin                 # pre-compiled dlib (Windows wheel, no compilation)
face_recognition_models
face_recognition         # the setup script installs this with "--no-deps" (to avoid recompiling dlib)
click
Pillow
opencv-python
pyserial

# ESP32 firmware flashing / file management tools
esptool                  # to flash MicroPython firmware to the board
mpremote                 # copy files to board, REPL, run tests
